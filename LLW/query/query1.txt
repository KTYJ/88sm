CREATE OR REPLACE VIEW v_order_type_trend AS
SELECT EXTRACT(YEAR FROM o.order_date)      AS sales_year,
       o.order_type,
       COUNT(DISTINCT o.order_id)           AS total_orders,
       SUM(oi.quantity * oi.unit_price)     AS revenue,
       ROUND(SUM(oi.quantity * oi.unit_price) /
             COUNT(DISTINCT o.order_id), 2) AS avg_order_value,
       COUNT(DISTINCT o.branch_id)          AS branches_served
FROM   orders o
JOIN   order_item oi ON o.order_id = oi.order_id
GROUP  BY EXTRACT(YEAR FROM o.order_date), o.order_type;

COLUMN item_id          FORMAT A10         HEADING 'ITEM ID';
COLUMN item_name        FORMAT A30 TRUNCATED HEADING 'ITEM NAME';
COLUMN category         FORMAT A14 TRUNCATED HEADING 'CATEGORY';
COLUMN total_units_sold FORMAT 999,990     HEADING 'UNITS SOLD';
COLUMN total_revenue    FORMAT 999,999,990.99 HEADING 'REVENUE (RM)';
COLUMN times_ordered    FORMAT 999,990     HEADING 'TIMES ORDERED';
COLUMN current_price    FORMAT 990.99      HEADING 'PRICE NOW';

SELECT v.item_id,
       v.item_name,
       v.category,
       v.total_units_sold,
       v.total_revenue,
       v.times_ordered,
       i.unit_price AS current_price
FROM   (SELECT *
        FROM   v_top_selling_items
        ORDER  BY total_units_sold DESC) v
JOIN   item i ON v.item_id = i.item_id
WHERE  ROWNUM <= 10;
