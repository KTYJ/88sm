CREATE OR REPLACE PROCEDURE rpt_category_sales_summary (
    p_category IN item.category%TYPE
) AS

    v_category    item.category%TYPE;
    v_cat_total   NUMBER(14,2);
    v_item_count  NUMBER;

    CURSOR itemCursor IS
        SELECT i.item_id,
               i.item_name,
               NVL(SUM(oi.quantity), 0) AS units,
               NVL(SUM(oi.quantity * oi.unit_price), 0) AS rev
        FROM item i
        LEFT JOIN order_item oi
            ON i.item_id = oi.item_id
        WHERE i.category = v_category
        GROUP BY i.item_id, i.item_name
        ORDER BY rev DESC;

BEGIN

    IF p_category IS NULL OR TRIM(p_category) IS NULL THEN
        RAISE_APPLICATION_ERROR(
            -20080,
            'Category is required.'
        );
    END IF;

    v_category := TRIM(p_category);

    DBMS_OUTPUT.PUT_LINE(
        '================================================================================'
    );

    DBMS_OUTPUT.PUT_LINE(
        'CATEGORY SALES SUMMARY REPORT'
    );

    DBMS_OUTPUT.PUT_LINE(
        '================================================================================'
    );

    DBMS_OUTPUT.PUT_LINE(
        'CATEGORY: ' || v_category
    );

    DBMS_OUTPUT.PUT_LINE(
        '================================================================================'
    );

    DBMS_OUTPUT.PUT_LINE(
        RPAD('Item', 34) ||
        LPAD('Units Sold', 12) ||
        LPAD('Revenue (RM)', 18)
    );

    DBMS_OUTPUT.PUT_LINE(
        '--------------------------------------------------------------------------------'
    );

    v_cat_total := 0;
    v_item_count := 0;

    FOR r_i IN itemCursor LOOP

        DBMS_OUTPUT.PUT_LINE(
            RPAD(r_i.item_name, 34) ||
            LPAD(TO_CHAR(r_i.units, '999,999'), 12) ||
            LPAD(TO_CHAR(r_i.rev, '999,999,990.00'), 18)
        );

        v_cat_total := v_cat_total + r_i.rev;
        v_item_count := v_item_count + 1;

    END LOOP;

    DBMS_OUTPUT.PUT_LINE(
        '--------------------------------------------------------------------------------'
    );

    DBMS_OUTPUT.PUT_LINE(
        'Category Total: RM ' ||
        TO_CHAR(v_cat_total, '999,999,990.00') ||
        '   |   Items: ' ||
        v_item_count
    );

    DBMS_OUTPUT.PUT_LINE(
        '================================================================================'
    );

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        RAISE_APPLICATION_ERROR(
            -20081,
            'No data found for category ' || v_category || '.'
        );
    WHEN VALUE_ERROR THEN
        RAISE_APPLICATION_ERROR(
            -20082,
            'Invalid category value in rpt_category_sales_summary: ' || SQLERRM
        );
    WHEN OTHERS THEN
        RAISE_APPLICATION_ERROR(
            -20991,
            'rpt_category_sales_summary failed: ' || SQLERRM
        );
END rpt_category_sales_summary;
/
