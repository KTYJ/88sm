CREATE OR REPLACE PROCEDURE sp_create_update_delivery (
    p_delivery_id   IN delivery.delivery_id%TYPE DEFAULT NULL,
    p_order_id      IN delivery.order_id%TYPE   DEFAULT NULL,
    p_company_id    IN delivery.company_id%TYPE DEFAULT NULL,
    p_address       IN delivery.delivery_address%TYPE DEFAULT NULL,
    p_new_status    IN delivery.delivery_status%TYPE DEFAULT 'Pending'
) AS
    v_current_status  delivery.delivery_status%TYPE;
    v_cnt             NUMBER;
    v_new_id          delivery.delivery_id%TYPE;
BEGIN
    -- Validation 1: status must be valid (both modes)
    IF p_new_status NOT IN ('Delivered', 'Out for Delivery',
                            'Pending', 'Preparing', 'Cancelled') THEN
        RAISE_APPLICATION_ERROR(-20050, 'Invalid delivery status value.');
    END IF;

    IF p_delivery_id IS NULL THEN
        -- ================= CREATE MODE =================

        -- Validation 2: order must exist
        SELECT COUNT(*) INTO v_cnt FROM orders WHERE order_id = p_order_id;
        IF v_cnt = 0 THEN
            RAISE_APPLICATION_ERROR(-20053, 'Order ' || p_order_id || ' does not exist.');
        END IF;

        -- Validation 3: one delivery per order (delivery.order_id is UNIQUE)
        SELECT COUNT(*) INTO v_cnt FROM delivery WHERE order_id = p_order_id;
        IF v_cnt > 0 THEN
            RAISE_APPLICATION_ERROR(-20054, 'Order ' || p_order_id ||
                ' already has a delivery record.');
        END IF;

        -- Validation 4: company must exist
        SELECT COUNT(*) INTO v_cnt FROM delivery_company WHERE company_id = p_company_id;
        IF v_cnt = 0 THEN
            RAISE_APPLICATION_ERROR(-20055, 'Company ' || p_company_id || ' does not exist.');
        END IF;

        -- Validation 5: address cannot be empty
        IF LENGTH(TRIM(p_address)) = 0 THEN
            RAISE_APPLICATION_ERROR(-20056, 'Delivery address cannot be empty.');
        END IF;

        -- Generate id (MAX+1 for the demo; production pattern uses
        -- seq_delivery_id from Task 8)
        SELECT NVL(MAX(delivery_id), 0) + 1 INTO v_new_id FROM delivery;

        INSERT INTO delivery (delivery_id, order_id, company_id,
                              delivery_address, delivery_date, delivery_status)
        VALUES (v_new_id, p_order_id, p_company_id, TRIM(p_address),
                CASE WHEN p_new_status = 'Delivered'
                     THEN SYSTIMESTAMP ELSE NULL END,   -- date only when Delivered
                p_new_status);

        DBMS_OUTPUT.PUT_LINE('Delivery ' || v_new_id ||
            ' created for order ' || p_order_id || ' [' || p_new_status || ']');
    ELSE
        -- ================= UPDATE MODE =================

        SELECT delivery_status INTO v_current_status
        FROM   delivery
        WHERE  delivery_id = p_delivery_id;

        -- Validation: delivered orders cannot be changed
        IF v_current_status = 'Delivered' THEN
            RAISE_APPLICATION_ERROR(-20051, 'Delivered orders cannot be modified.');
        END IF;

        -- Status update + date rule:
        --   -> Delivered : set delivery_date = SYSTIMESTAMP
        --   -> Cancelled : clear delivery_date
        --   -> otherwise : keep existing date
        UPDATE delivery
        SET    delivery_status = p_new_status,
               delivery_date = CASE
                   WHEN p_new_status = 'Delivered' THEN SYSTIMESTAMP
                   WHEN p_new_status = 'Cancelled'  THEN NULL
                   ELSE delivery_date
               END
        WHERE  delivery_id = p_delivery_id;

        DBMS_OUTPUT.PUT_LINE('Delivery ' || p_delivery_id ||
            ' status updated: ' || v_current_status || ' -> ' || p_new_status);
    END IF;

    COMMIT;
EXCEPTION
    WHEN NO_DATA_FOUND THEN
        RAISE_APPLICATION_ERROR(-20052, 'Delivery record not found.');
    WHEN OTHERS THEN
        IF SQLCODE BETWEEN -20999 AND -20000 THEN
            RAISE;
        ELSE
            RAISE_APPLICATION_ERROR(-20906, 'sp_create_update_delivery failed: ' || SQLERRM);
        END IF;
END;
