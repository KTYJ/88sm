CREATE OR REPLACE PROCEDURE sp_update_item_price (
    p_item_id   IN item.item_id%TYPE,
    p_new_price IN item.unit_price%TYPE
)
IS
    v_old_price item.unit_price%TYPE;
    v_latest_cost procurement.unit_cost%TYPE;

BEGIN

    /* ==========================================
       1. Get Existing Price
       ========================================== */

    SELECT unit_price
    INTO v_old_price
    FROM item
    WHERE item_id = p_item_id;


    /* ==========================================
       2. Validate New Price
       ========================================== */

    IF p_new_price < 0 THEN

        RAISE_APPLICATION_ERROR(
            -20010,
            'Selling price cannot be negative.'
        );

    END IF;


    /* ==========================================
       3. Get Latest Procurement Cost
       ========================================== */

    BEGIN

        SELECT unit_cost
        INTO v_latest_cost

        FROM (
            SELECT
                unit_cost

            FROM procurement

            WHERE item_id = p_item_id

            ORDER BY procurement_date DESC
        )

        WHERE ROWNUM = 1;


    EXCEPTION

        WHEN NO_DATA_FOUND THEN

            v_latest_cost := NULL;

    END;


    /* ==========================================
       4. Compare Selling Price with Cost
       ========================================== */

    IF v_latest_cost IS NOT NULL
       AND p_new_price < v_latest_cost THEN

        RAISE_APPLICATION_ERROR(
            -20011,
            'New selling price cannot be lower than latest procurement cost of RM' ||
            TO_CHAR(
                v_latest_cost,
                'FM999,999,990.00'
            )
        );

    END IF;


    /* ==========================================
       5. Update Price
       ========================================== */

    UPDATE item

    SET unit_price = p_new_price

    WHERE item_id = p_item_id;


    /* ==========================================
       6. Display Result
       ========================================== */

    DBMS_OUTPUT.PUT_LINE(
        '=========================================='
    );

    DBMS_OUTPUT.PUT_LINE(
        'ITEM PRICE UPDATED'
    );

    DBMS_OUTPUT.PUT_LINE(
        '=========================================='
    );

    DBMS_OUTPUT.PUT_LINE(
        'Item ID   : ' || p_item_id
    );

    DBMS_OUTPUT.PUT_LINE(
        'Old Price : RM' ||
        TO_CHAR(
            v_old_price,
            'FM999,999,990.00'
        )
    );

    DBMS_OUTPUT.PUT_LINE(
        'New Price : RM' ||
        TO_CHAR(
            p_new_price,
            'FM999,999,990.00'
        )
    );

    DBMS_OUTPUT.PUT_LINE(
        '=========================================='
    );


EXCEPTION

    WHEN NO_DATA_FOUND THEN

        RAISE_APPLICATION_ERROR(
            -20012,
            'Item ID ' || p_item_id || ' does not exist.'
        );


    WHEN OTHERS THEN

        IF SQLCODE BETWEEN -20099 AND -20000 THEN

            RAISE;

        ELSE

            RAISE_APPLICATION_ERROR(
                -20013,
                'Unexpected error: ' || SQLERRM
            );

        END IF;

END;
