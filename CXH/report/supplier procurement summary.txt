CREATE OR REPLACE PROCEDURE rpt_supplier_procurement
IS

    CURSOR c_supplier IS

        SELECT
            supplier_id,
            supplier_name,
            total_items_supplied,
            branches_supplied,
            total_quantity_supplied,
            total_procurement_cost

        FROM vw_supplier_performance

        ORDER BY total_procurement_cost DESC;


    v_rank        NUMBER := 0;
    v_grand_total NUMBER := 0;

BEGIN

    /* =====================================================
       REPORT HEADER
       ===================================================== */

    DBMS_OUTPUT.PUT_LINE(
        RPAD('=', 110, '=')
    );

    DBMS_OUTPUT.PUT_LINE(
        '                    88 SPEEDMART'
    );

    DBMS_OUTPUT.PUT_LINE(
        '              SUPPLIER PROCUREMENT SUMMARY'
    );

    DBMS_OUTPUT.PUT_LINE(
        'Generated: ' ||
        TO_CHAR(SYSDATE, 'DD-MON-YYYY HH24:MI')
    );

    DBMS_OUTPUT.PUT_LINE(
        RPAD('=', 110, '=')
    );


    /* =====================================================
       TABLE HEADER
       ===================================================== */

    DBMS_OUTPUT.PUT_LINE(

        RPAD('Rank', 7) ||

        RPAD('Supplier', 30) ||

        LPAD('Items', 10) ||

        LPAD('Branches', 12) ||

        LPAD('Quantity', 16) ||

        LPAD('Cost (RM)', 20)

    );


    DBMS_OUTPUT.PUT_LINE(
        RPAD('-', 110, '-')
    );


    /* =====================================================
       SUPPLIER DATA
       ===================================================== */

    FOR r IN c_supplier LOOP

        v_rank := v_rank + 1;

        v_grand_total :=
            v_grand_total +
            NVL(r.total_procurement_cost, 0);


        DBMS_OUTPUT.PUT_LINE(

            RPAD(
                TO_CHAR(v_rank),
                7
            ) ||

            RPAD(
                SUBSTR(
                    r.supplier_name,
                    1,
                    28
                ),
                30
            ) ||

            LPAD(
                TO_CHAR(
                    NVL(r.total_items_supplied, 0)
                ),
                10
            ) ||

            LPAD(
                TO_CHAR(
                    NVL(r.branches_supplied, 0)
                ),
                12
            ) ||

            LPAD(
                TO_CHAR(
                    NVL(r.total_quantity_supplied, 0)
                ),
                16
            ) ||

            LPAD(
                TO_CHAR(
                    NVL(r.total_procurement_cost, 0),
                    'FM999,999,990.00'
                ),
                20
            )

        );

    END LOOP;


    /* =====================================================
       TOTAL
       ===================================================== */

    DBMS_OUTPUT.PUT_LINE(
        RPAD('-', 110, '-')
    );

    DBMS_OUTPUT.PUT_LINE(

        RPAD(
            'TOTAL PROCUREMENT COST',
            90
        ) ||

        LPAD(
            'RM' ||
            TO_CHAR(
                v_grand_total,
                'FM999,999,990.00'
            ),
            20
        )

    );

    DBMS_OUTPUT.PUT_LINE(
        RPAD('=', 110, '=')
    );


EXCEPTION

    WHEN OTHERS THEN

        DBMS_OUTPUT.PUT_LINE(
            'Error generating report: ' ||
            SQLERRM
        );

END;
/

SET LINESIZE 150
SET PAGESIZE 50
SET SERVEROUTPUT ON
EXEC rpt_supplier_procurement;
