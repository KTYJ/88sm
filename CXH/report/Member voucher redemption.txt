CREATE OR REPLACE PROCEDURE rpt_member_voucher_redemption (
    p_member_id IN member.member_id%TYPE
)
IS

    v_member_name     member.name%TYPE;
    v_total_spending  NUMBER;

    /* ==========================================
       OUTER CURSOR - ORDERS
       ========================================== */

    CURSOR c_orders IS

        SELECT
            o.order_id,
            o.order_date,
            o.order_type

        FROM orders o

        WHERE o.member_id = p_member_id

        ORDER BY
            o.order_date DESC;


    /* ==========================================
       INNER CURSOR - VOUCHER REDEMPTIONS
       ========================================== */

    CURSOR c_vouchers (
        p_order_id orders.order_id%TYPE
    ) IS

        SELECT
            r.redemption_id,
            r.voucher_id,
            v.voucher_type,
            v.voucher_value,
            r.redeemed_date

        FROM redemption r

        JOIN voucher v
            ON r.voucher_id = v.voucher_id

        WHERE r.order_id = p_order_id

        ORDER BY
            r.redeemed_date;


    v_found BOOLEAN := FALSE;


BEGIN

    /* ==========================================
       GET MEMBER INFORMATION
       ========================================== */

    SELECT
        name

    INTO
        v_member_name

    FROM member

    WHERE member_id = p_member_id;


    /* ==========================================
       CALCULATE TOTAL SPENDING
       ========================================== */

    v_total_spending :=
        fn_member_total_spending(p_member_id);


    /* ==========================================
       REPORT HEADER
       ========================================== */

    DBMS_OUTPUT.PUT_LINE(
        RPAD('=', 140, '=')
    );

    DBMS_OUTPUT.PUT_LINE(
        '88 SPEEDMART - MEMBER VOUCHER REDEMPTION REPORT'
    );

    DBMS_OUTPUT.PUT_LINE('');

    DBMS_OUTPUT.PUT_LINE(
        'Member ID      : ' || p_member_id
    );

    DBMS_OUTPUT.PUT_LINE(
        'Member Name    : ' || v_member_name
    );

    DBMS_OUTPUT.PUT_LINE(
        'Total Spending : RM' ||
        TO_CHAR(
            v_total_spending,
            'FM999,999,990.00'
        )
    );

    DBMS_OUTPUT.PUT_LINE('');

    DBMS_OUTPUT.PUT_LINE(
        RPAD('=', 140, '=')
    );


    /* ==========================================
       COLUMN HEADER
       ========================================== */

    DBMS_OUTPUT.PUT_LINE(

        RPAD('Order ID', 11) ||

        RPAD('Order Date', 22) ||

        RPAD('Order Type', 16) ||

        RPAD('Voucher', 10) ||

        RPAD('Voucher Type', 18) ||

        LPAD('Voucher Value', 15)

    );


    DBMS_OUTPUT.PUT_LINE(
        RPAD('-', 92, '-')
    );


    /* ==========================================
       REPORT BODY
       ========================================== */

    FOR r_order IN c_orders LOOP

        FOR r_voucher IN c_vouchers(
            r_order.order_id
        ) LOOP

            v_found := TRUE;

            DBMS_OUTPUT.PUT_LINE(

                /* Order ID */
                RPAD(
                    TO_CHAR(
                        r_order.order_id
                    ),
                    11
                ) ||

                /* Order Date */
                RPAD(
                    TO_CHAR(
                        r_order.order_date,
                        'DD-MON-YYYY HH24:MI'
                    ),
                    22
                ) ||

                /* Order Type */
                RPAD(
                    r_order.order_type,
                    16
                ) ||

                /* Voucher ID */
                RPAD(
                    TO_CHAR(
                        r_voucher.voucher_id
                    ),
                    10
                ) ||

                /* Voucher Type */
                RPAD(
                    r_voucher.voucher_type,
                    18
                ) ||

                /* Voucher Value */
                LPAD(

                    CASE

                        WHEN r_voucher.voucher_type = 'Amount'

                        THEN
                            'RM' ||
                            TO_CHAR(
                                r_voucher.voucher_value,
                                'FM999,990.00'
                            )

                        ELSE
                            TO_CHAR(
                                r_voucher.voucher_value,
                                'FM990.00'
                            ) || '%'

                    END,

                    15
                )

            );

        END LOOP;

    END LOOP;


    /* ==========================================
       NO RECORD MESSAGE
       ========================================== */

    IF NOT v_found THEN

        DBMS_OUTPUT.PUT_LINE('');

        DBMS_OUTPUT.PUT_LINE(
            'No voucher redemption records found.'
        );

    END IF;


    /* ==========================================
       REPORT FOOTER
       ========================================== */

    DBMS_OUTPUT.PUT_LINE('');

    DBMS_OUTPUT.PUT_LINE(
        RPAD('=', 140, '=')
    );


EXCEPTION

    /* ==========================================
       MEMBER DOES NOT EXIST
       ========================================== */

    WHEN NO_DATA_FOUND THEN

        RAISE_APPLICATION_ERROR(
            -20030,
            'Member ID ' ||
            p_member_id ||
            ' does not exist.'
        );


    /* ==========================================
       OTHER ERRORS
       ========================================== */

    WHEN OTHERS THEN

        DBMS_OUTPUT.PUT_LINE(
            'Error generating report: ' ||
            SQLERRM
        );

END;
/

SET LINESIZE 150
SET PAGESIZE 50
SET SERVEROUTPUT ON
EXEC rpt_member_voucher_redemption(6014);
