SELECT
    object_name,
    object_type,
    status
FROM user_objects
WHERE object_name IN (
    'VW_SUPPLIER_PERFORMANCE',
    'VW_MEMBER_PURCHASE_BEHAVIOR',
    'SEQ_MEMBER_ID',
    'SP_REGISTER_MEMBER',
    'SP_UPDATE_ITEM_PRICE',
    'TRG_VALIDATE_ITEM_PRICE',
    'TRG_PREVENT_NEGATIVE_POINTS',
    'RPT_SUPPLIER_PROCUREMENT',
    'RPT_MEMBER_VOUCHER_REDEMPTION',
    'FN_MEMBER_TOTAL_SPENDING'
)
ORDER BY object_type, object_name;
