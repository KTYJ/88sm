CREATE OR REPLACE VIEW vw_supplier_performance AS
SELECT
    s.supplier_id,
    s.supplier_name,

    COUNT(DISTINCT p.item_id) AS total_items_supplied,

    COUNT(DISTINCT p.branch_id) AS branches_supplied,

    SUM(p.quantity) AS total_quantity_supplied,

    SUM(p.quantity * p.unit_cost) AS total_procurement_cost,

    AVG(p.unit_cost) AS average_unit_cost

FROM supplier s

JOIN procurement p
    ON s.supplier_id = p.supplier_id

JOIN item i
    ON p.item_id = i.item_id

JOIN branch b
    ON p.branch_id = b.branch_id

GROUP BY
    s.supplier_id,
    s.supplier_name;
/
