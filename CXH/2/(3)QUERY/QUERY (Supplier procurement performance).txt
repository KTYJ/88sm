SELECT
    supplier_id,
    supplier_name,

    total_items_supplied,
    branches_supplied,
    total_quantity_supplied,

    TO_CHAR(
        total_procurement_cost,
        'FM999,999,990.00'
    ) AS total_procurement_cost,

    TO_CHAR(
        average_unit_cost,
        'FM999,999,990.00'
    ) AS average_unit_cost,

    RANK() OVER (
        ORDER BY total_procurement_cost DESC
    ) AS procurement_rank

FROM vw_supplier_performance

ORDER BY
    total_procurement_cost DESC;
