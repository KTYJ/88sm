CL SCR
SET SERVEROUTPUT ON
SET DEFINE OFF
SET LINESIZE 120;

-- =============================================================================
-- REPORT 2: MEMBER POINT ACTIVITY REPORT
-- By: Ryan
-- Cursor: Nested cursor
-- Outer cursor  : Members
-- Inner cursor  : Point transactions for each member
-- Parameter     : Start Date and End Date
-- =============================================================================


CREATE OR REPLACE PROCEDURE rp_member_point_activity (
    p_start_date IN DATE,
    p_end_date   IN DATE
) IS

    -- =========================================================================
    -- OUTER CURSOR
    -- Gets members who have point transactions in the selected period.
    -- =========================================================================

    CURSOR c_member IS
        SELECT DISTINCT
            m.member_id,
            m.name,
            m.points_balance
        FROM member m
        JOIN point_history ph
            ON m.member_id = ph.member_id
        WHERE ph.redemption_date >= p_start_date
          AND ph.redemption_date < p_end_date + 1
        ORDER BY m.member_id;


    -- =========================================================================
    -- INNER CURSOR
    -- Gets point transactions belonging to the current member.
    -- =========================================================================

    CURSOR c_point_history (
        p_member_id member.member_id%TYPE
    ) IS
        SELECT
            ph.point_redemption_id,
            ph.transaction_type,
            ph.amount,
            ph.redemption_date
        FROM point_history ph
        WHERE ph.member_id = p_member_id
          AND ph.redemption_date >= p_start_date
          AND ph.redemption_date < p_end_date + 1
        ORDER BY ph.redemption_date;


    -- =========================================================================
    -- OUTER CURSOR VARIABLES
    -- =========================================================================

    v_member_id       member.member_id%TYPE;
    v_member_name     member.name%TYPE;
    v_points_balance  member.points_balance%TYPE;


    -- =========================================================================
    -- MEMBER SUMMARY VARIABLES
    -- =========================================================================

    v_earned         NUMBER := 0;
    v_used           NUMBER := 0;
    v_net_activity   NUMBER := 0;
    v_transaction_count NUMBER := 0;

    v_total_members  NUMBER := 0;
    v_total_earned   NUMBER := 0;
    v_total_used     NUMBER := 0;


    -- =========================================================================
    -- REPORT CONTROL
    -- =========================================================================

    v_found BOOLEAN := FALSE;


BEGIN

    -- =========================================================================
    -- VALIDATE DATE RANGE
    -- =========================================================================

    IF p_start_date > p_end_date THEN

        RAISE_APPLICATION_ERROR(
            -20041,
            'Invalid date range: Start date cannot be later than end date.'
        );

    END IF;


    -- =========================================================================
    -- REPORT HEADER
    -- =========================================================================

    DBMS_OUTPUT.PUT_LINE(
        '========================================================================'
    );

    DBMS_OUTPUT.PUT_LINE(
        '                     MEMBER POINT ACTIVITY REPORT'
    );

    DBMS_OUTPUT.PUT_LINE(
        '========================================================================'
    );

    DBMS_OUTPUT.PUT_LINE(
        'Period      : ' ||
        TO_CHAR(p_start_date, 'DD-MON-YYYY') ||
        ' to ' ||
        TO_CHAR(p_end_date, 'DD-MON-YYYY')
    );

    DBMS_OUTPUT.PUT_LINE(
        'Generated On: ' ||
        TO_CHAR(SYSDATE, 'DD-MON-YYYY HH24:MI')
    );

    DBMS_OUTPUT.PUT_LINE(
        '========================================================================'
    );



    -- =========================================================================
    -- OUTER CURSOR
    -- =========================================================================

    OPEN c_member;

    LOOP

        FETCH c_member
        INTO
            v_member_id,
            v_member_name,
            v_points_balance;

        EXIT WHEN c_member%NOTFOUND;

        v_found := TRUE;
        v_total_members := v_total_members + 1;

        -- Reset member totals
        v_earned := 0;
        v_used := 0;
        v_net_activity := 0;
        v_transaction_count := 0;


        -- =====================================================================
        -- MEMBER INFORMATION
        -- =====================================================================

        DBMS_OUTPUT.PUT_LINE(CHR(10));

        DBMS_OUTPUT.PUT_LINE(
            'Member ID      : ' || v_member_id
        );

        DBMS_OUTPUT.PUT_LINE(
            'Member Name    : ' || SUBSTR(v_member_name, 1, 40)
        );

        DBMS_OUTPUT.PUT_LINE(
            'Current Balance: ' || v_points_balance || ' points'
        );

        DBMS_OUTPUT.PUT_LINE(
            '------------------------------------------------------------------------'
        );

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Point ID', 14) ||
            RPAD('Date', 15) ||
            RPAD('Type', 12) ||
            LPAD('Points', 10)
        );

        DBMS_OUTPUT.PUT_LINE(
            RPAD('-', 14, '-') ||
            RPAD('-', 15, '-') ||
            RPAD('-', 12, '-') ||
            LPAD('-', 10, '-')
        );


        -- =====================================================================
        -- INNER CURSOR
        -- Process each point transaction for the current member.
        -- =====================================================================

        FOR r_point IN c_point_history(v_member_id)
        LOOP

            DBMS_OUTPUT.PUT_LINE(
                RPAD(r_point.point_redemption_id, 14) ||
                RPAD(
                    TO_CHAR(r_point.redemption_date, 'DD-MON-YYYY'),
                    15
                ) ||
                RPAD(r_point.transaction_type, 12) ||
                LPAD(
                    TO_CHAR(r_point.amount),
                    10
                )
            );


            -- ================================================================
            -- Calculate member totals
            -- ================================================================

            IF r_point.transaction_type = 'Earned' THEN

                v_earned := v_earned + r_point.amount;

            ELSIF r_point.transaction_type = 'Used' THEN

                v_used := v_used + r_point.amount;

            END IF;

            v_transaction_count :=
                v_transaction_count + 1;

        END LOOP;


        -- =====================================================================
        -- MEMBER SUMMARY
        -- =====================================================================

        v_net_activity := v_earned - v_used;

        v_total_earned :=
            v_total_earned + v_earned;

        v_total_used :=
            v_total_used + v_used;


        DBMS_OUTPUT.PUT_LINE(
            '------------------------------------------------------------------------'
        );

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Points Earned', 25) ||
            LPAD(TO_CHAR(v_earned), 12)
        );

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Points Used', 25) ||
            LPAD(TO_CHAR(v_used), 12)
        );

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Net Point Activity', 25) ||
            LPAD(TO_CHAR(v_net_activity), 12)
        );

        DBMS_OUTPUT.PUT_LINE(
            RPAD('Transactions', 25) ||
            LPAD(TO_CHAR(v_transaction_count), 12)
        );

        DBMS_OUTPUT.PUT_LINE(
            '========================================================================'
        );

    END LOOP;


    -- =========================================================================
    -- CLOSE OUTER CURSOR
    -- =========================================================================

    CLOSE c_member;


    -- =========================================================================
    -- OVERALL MANAGEMENT SUMMARY
    -- =========================================================================

    IF v_found THEN

        DBMS_OUTPUT.PUT_LINE(CHR(10));

        DBMS_OUTPUT.PUT_LINE(
            '========================================================================'
        );

        DBMS_OUTPUT.PUT_LINE(
            '                       MANAGEMENT SUMMARY'
        );

        DBMS_OUTPUT.PUT_LINE(
            '========================================================================'
        );

        DBMS_OUTPUT.PUT_LINE(
            'Members with Point Activity : ' ||
            v_total_members
        );

        DBMS_OUTPUT.PUT_LINE(
            'Total Points Earned         : ' ||
            v_total_earned
        );

        DBMS_OUTPUT.PUT_LINE(
            'Total Points Used           : ' ||
            v_total_used
        );

        DBMS_OUTPUT.PUT_LINE(
            'Net Point Activity          : ' ||
            (v_total_earned - v_total_used)
        );

        DBMS_OUTPUT.PUT_LINE(
            '========================================================================'
        );

    ELSE

        DBMS_OUTPUT.PUT_LINE(CHR(10));

        DBMS_OUTPUT.PUT_LINE(
            'No point activity found for the selected period.'
        );

        DBMS_OUTPUT.PUT_LINE(
            '========================================================================'
        );

    END IF;


    -- =========================================================================
    -- END OF REPORT
    -- =========================================================================

    DBMS_OUTPUT.PUT_LINE(
        '*** End of Report ***'
    );

    DBMS_OUTPUT.PUT_LINE(
        '========================================================================'
    );


EXCEPTION

    -- =========================================================================
    -- DEFAULT EXCEPTION
    -- =========================================================================

    WHEN OTHERS THEN

        IF c_member%ISOPEN THEN
            CLOSE c_member;
        END IF;

        DBMS_OUTPUT.PUT_LINE(
            'Error: ' || SQLERRM
        );

END;
