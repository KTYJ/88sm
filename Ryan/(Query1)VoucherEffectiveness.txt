-- ============================================================
-- QUERY 1: Voucher Redemption Effectiveness
-- Period: Last 2 Years
-- ============================================================


-- ============================================================
-- VIEW 1: VOUCHER REDEMPTION EFFECTIVENESS
-- ============================================================

CREATE OR REPLACE VIEW voucher_redemption_view AS
SELECT
    r.voucher_id,
    COUNT(DISTINCT r.redemption_id) AS redemption_count,
    COUNT(DISTINCT r.order_id) AS order_count,
    SUM(oi.quantity * oi.unit_price) AS total_sales
FROM redemption r
JOIN orders o
    ON r.order_id = o.order_id
JOIN order_item oi
    ON o.order_id = oi.order_id
WHERE o.order_date >= ADD_MONTHS(SYSDATE, -24)
GROUP BY r.voucher_id
HAVING COUNT(DISTINCT r.redemption_id) >= 10;


-- ============================================================
-- SQL*PLUS OUTPUT FORMATTING
-- ============================================================

SET LINESIZE 120
SET PAGESIZE 30
SET VERIFY OFF
SET UNDERLINE '='
SET DEFINE ON

COLUMN rank FORMAT 99 HEADING 'Rank'
COLUMN voucher_id FORMAT A10 HEADING 'Voucher ID'
COLUMN voucher_type FORMAT A15 HEADING 'Voucher Type'
COLUMN voucher_value FORMAT 999.99 HEADING 'Value'
COLUMN redemption_count FORMAT 99,999 HEADING 'Redemptions'
COLUMN total_sales FORMAT 999,999.99 HEADING 'Total Sales (RM)'
COLUMN avg_order_value FORMAT 999,999.99 HEADING 'Avg Order (RM)'


-- ============================================================
-- GET REPORT DATE RANGE
-- ============================================================

COLUMN start_date NEW_VALUE v_start_date NOPRINT
COLUMN end_date NEW_VALUE v_end_date NOPRINT

SELECT
    TO_CHAR(ADD_MONTHS(SYSDATE, -24), 'DD-MON-YYYY') AS start_date,
    TO_CHAR(SYSDATE, 'DD-MON-YYYY') AS end_date
FROM dual;


-- ============================================================
-- REPORT TITLE
-- ============================================================

TTITLE CENTER 'Voucher Redemption Effectiveness Report' SKIP 1 -
       CENTER '&v_start_date to &v_end_date' SKIP 1 -
       CENTER 'Top 10 Most Redeemed Vouchers' SKIP 1 -
       RIGHT 'Page: ' FORMAT 999 SQL.PNO SKIP 2

BTITLE CENTER '----------------------------------------------' SKIP 1 -
       CENTER '*** End of Report ***' SKIP 1


-- ============================================================
-- QUERY 1
-- ============================================================

SELECT
    ROWNUM AS rank,
    v.voucher_id,
    v.voucher_type,
    v.voucher_value,
    v.redemption_count,
    v.total_sales,
    ROUND(v.total_sales / v.order_count, 2) AS avg_order_value
FROM
(
    SELECT
        vo.voucher_id,
        vo.voucher_type,
        vo.voucher_value,
        vr.redemption_count,
        vr.order_count,
        vr.total_sales
    FROM voucher vo
    JOIN voucher_redemption_view vr
        ON vo.voucher_id = vr.voucher_id
    ORDER BY
        vr.redemption_count DESC,
        vr.total_sales DESC
) v
WHERE ROWNUM <= 10;


-- ============================================================
-- CLEAN UP SQL*PLUS SETTINGS
-- ============================================================

TTITLE OFF
BTITLE OFF

CLEAR COLUMNS
CLEAR BREAKS
CLEAR COMPUTES

SET UNDERLINE '-'
SET DEFINE OFF;


-- ============================================================
-- DROP VIEW AFTER EXECUTION
-- ============================================================

DROP VIEW voucher_redemption_view;
